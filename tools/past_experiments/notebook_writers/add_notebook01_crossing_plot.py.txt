def main() -> None:
    notebook = nbf.read(TARGET, as_version=4)
    existing = [index for index, cell in enumerate(notebook.cells) if cell.cell_type == "markdown" and cell.source.startswith(HEADING)]
    if existing:
        index = existing[0]
        if len(existing) != 1 or notebook.cells[index + 1].cell_type != "code":
            raise ValueError("Unexpected existing crossing section")
        notebook.cells[index].source = MARKDOWN
        notebook.cells[index + 1].source = CODE.strip() + "\n"
        notebook.cells[index + 1].outputs = []
        notebook.cells[index + 1].execution_count = None
    else:
        scope = [index for index, cell in enumerate(notebook.cells) if cell.cell_type == "markdown" and cell.source.startswith("## Scope")]
        if len(scope) != 1:
            raise ValueError("Expected exactly one scope section")
        notebook.cells[scope[0]:scope[0]] = [nbf.v4.new_markdown_cell(MARKDOWN), nbf.v4.new_code_cell(CODE.strip() + "\n")]
    nbf.validate(notebook)
    nbf.write(notebook, TARGET)
    print(f"Updated {TARGET} ({len(notebook.cells)} cells)")


if __name__ == "__main__":
    main()
